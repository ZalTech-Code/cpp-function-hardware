/*
 * 09_nonblocking_cloud.ino — full source. The function running on the board
 * right now is highlighted below. (Mirror of examples/09_nonblocking_cloud.ino)
 */
// ---- Uno R3 board config (inlined from board_config.h) ----
const int   LED_PIN    = LED_BUILTIN;  // pin 13 on Uno R3
const int   SENSOR_PIN = A0;
const int   ADC_MAX    = 1023;
const char* BOARD_NAME = "Arduino Uno R3"
const int LED = LED_PIN;

// ---- Blink state (non-blocking) ----
bool          ledState      = false;
unsigned long lastBlink     = 0;
int           blinkInterval = 500;  // ms between toggles
int           blinkTarget   = 0;    // toggles remaining (0 = idle)
int           blinkCount    = 0;    // toggles done this sequence

// ---- Status reporting / flow ----
const int     STATUS_INTERVAL = 200;
unsigned long lastStatus      = 0;
unsigned long loopIterations  = 0;   // how many times loop() has run

// ---- Trace mode (dashboard-controlled) ----
bool          traceMode   = false;
int           tracePaceMs = 400;     // ms per loop() while tracing
unsigned long lastTraceLoop = 0;

void setup() {
  pinMode(LED, OUTPUT);
  Serial.begin(9600);
  Serial.print("BOARD;name="); Serial.print(BOARD_NAME);
  Serial.print(";adcmax=");    Serial.println(ADC_MAX);
  Serial.println("READY;sketch=09_nonblocking_cloud");
}

void loop() {
  unsigned long now = millis();
  loopIterations++;

  // In trace mode, run at a watchable pace and announce each pass.
  bool tracing = false;
  if (traceMode && (now - lastTraceLoop >= (unsigned long)tracePaceMs)) {
    tracing = true;
    lastTraceLoop = now;
    flowEventKV("loop", "loop", "iter", (long)loopIterations);
  }

  // 1) LED blink state machine (the "LED loop" inside the main loop).
  if (tracing) flowEvent("enter", "updateBlink");
  updateBlink(now);
  if (tracing) flowEvent("exit", "updateBlink");

  // 2) Periodic status report for the dashboard.
  if (now - lastStatus >= STATUS_INTERVAL) {
    reportStatus(now);
    lastStatus = now;
  }

  // 3) Handle dashboard commands (never blocks the loop).
  if (Serial.available()) {
    String line = Serial.readStringUntil('\n');
    line.trim();
    if (line.startsWith("CMD;")) handleCommand(line);
  }

  // Pace the loop while tracing so each pass is individually visible.
  if (traceMode) delay(tracePaceMs);
}

// ---- FUNCTIONS ----

// Emit "FLOW;evt=...;fn=..." — one line the dashboard renders.
void flowEvent(const char* evt, const char* fn) {
  Serial.print("FLOW;evt="); Serial.print(evt);
  Serial.print(";fn=");      Serial.println(fn);
}

void flowEventKV(const char* evt, const char* fn, const char* key, long val) {
  Serial.print("FLOW;evt="); Serial.print(evt);
  Serial.print(";fn=");      Serial.print(fn);
  Serial.print(';');         Serial.print(key);
  Serial.print('=');         Serial.println(val);
}

// The "LED loop": toggles the LED on schedule. Called every loop() iteration.
void updateBlink(unsigned long now) {
  if (blinkTarget <= 0) return;                        // idle: no blink
  if (now - lastBlink < (unsigned long)blinkInterval) return; // not time yet

  ledState = !ledState;
  digitalWrite(LED, ledState ? HIGH : LOW);
  lastBlink = now;
  blinkTarget--;
  blinkCount++;

  if (blinkTarget == 0) {          // sequence finished: leave LED off
    ledState = false;
    digitalWrite(LED, LOW);
  }

  // One FLOW line per toggle (every toggle when tracing).
  if (blinkInterval >= 100 || traceMode) {
    flowEventKV("call", "updateBlink", "led", ledState ? 1 : 0);
  }
}

// One STATUS line the dashboard parses to update its cards.
void reportStatus(unsigned long now) {
  flowEventKV("enter", "reportStatus", "iter", (long)loopIterations);
  Serial.print("STATUS;uptime="); Serial.print(now);
  Serial.print(";led=");          Serial.print(ledState ? 1 : 0);
  Serial.print(";blinks=");       Serial.print(blinkCount);
  Serial.print(";sensor=0");
  Serial.println();
  flowEvent("exit", "reportStatus");
}

// Parse "CMD;key=value;key=value" and act on each pair.
void handleCommand(String line) {
  String safe = line;
  safe.replace(';', ' ');
  Serial.print("FLOW;evt=enter;fn=handleCommand;cmd=");
  Serial.println(safe);

  bool needStatus = false;
  int  start = 4;  // skip "CMD;"

  while (start < (int)line.length()) {
    int sep = line.indexOf(';', start);
    if (sep < 0) sep = line.length();
    String token = line.substring(start, sep);
    int eq = token.indexOf('=');
    if (eq > 0) {
      String key = token.substring(0, eq);
      String val = token.substring(eq + 1);

      if (key == "led") {
        ledState = (val == "1" || val == "true");
        digitalWrite(LED, ledState ? HIGH : LOW);
        blinkTarget = 0;
        flowEventKV("call", "setLed", "val", ledState ? 1 : 0);
        needStatus = true;

      } else if (key == "blink") {
        int n = val.toInt();
        if (n > 0) {
          ledState = false;
          digitalWrite(LED, LOW);
          blinkTarget = n * 2;                 // each blink = on + off
          blinkCount  = 0;
          lastBlink   = millis() - blinkInterval;
        } else {
          ledState = false;
          digitalWrite(LED, LOW);
          blinkTarget = 0;
        }
        flowEventKV("call", "startBlink", "val", n);
        needStatus = true;

      } else if (key == "interval") {
        int ms = val.toInt();
        if (ms >= 50 && ms <= 2000) {
          blinkInterval = ms;
          flowEventKV("call", "setInterval", "val", ms);
        }

      } else if (key == "trace") {
        traceMode = (val == "1" || val == "on" || val == "true");
        lastTraceLoop = millis();
        flowEventKV("call", "setTrace", "val", traceMode ? 1 : 0);
        needStatus = true;

      } else if (key == "pace") {
        int ms = val.toInt();
        if (ms >= 50 && ms <= 2000) {
          tracePaceMs = ms;
          flowEventKV("call", "setPace", "val", ms);
        }

      } else if (key == "report") {
        needStatus = true;
      }
    }
    start = sep + 1;
  }

  flowEvent("exit", "handleCommand");
  if (needStatus) reportStatus(millis());
}
